class _PromptContext:
    """
    Represents the injected context generated from a registered prompt.
    Attributes:
        prompt_index: Index of the prompt
        relevant: Whether the prompt is highly relevant to <goal specified by the user> based on general understanding of the goal,
        if the prompt is not highly relevant then the value is False,otherwise
        it is True. Prompts that output JSON or diagrams should be marked as not relevant.
    """
    prompt_index: int
    relevant: bool = False
    def __str__(self):
        if self.relevant:
            context_str = "{{answers[" + str(self.prompt_index) + "]}}""
            return (
            f"--- BEGIN ADDITIONAL CONTEXT: ITEM {self.prompt_index} ---\n"
            f"{context_str}\n"
            f"--- END ADDITIONAL CONTEXT: ITEM {self.prompt_index} ---\n\n"
           )
        else:
            return ""

class _CustomReportSection:
    """
    Represents a custom report section.
    Attributes:
        section_title: Appropriate title of the section given the user_prompt
        system_prompt: Appropriate system prompt of the section given the user_prompt
        context: Context for the section
        user_prompt: User prompt for the section given the goal
        goal: Goal for this section
        top3_context_list_as_str: Context list as a string
    """
    section_title: str
    system_prompt: str
    context_list: PromptContext[]
    user_prompt: str
    number_of_prompts: int
    goal: str
    @property
    def top3_context_list_as_str(self) -> str:
        """top 3 elements of context_list concatenated as a string"""
        ctxs = [ctx for ctx in self.context_list if ctx.relevant]
        return "\n".join(str(ctx) for ctx in ctxs[:3])
    def custom_report_prompt_name(self) -> str:
        """custom report prompt name"""
        return f"""{self.number_of_prompts}.txt"
